\documentclass[border=4mm]{standalone}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usepgfplotslibrary{groupplots,fillbetween}

% Okabe-Ito colours
\definecolor{oiGreen}{HTML}{009E73}   % theoretical
\definecolor{oiBlue}{HTML}{0072B2}    % Monte Carlo mean
\definecolor{oiSky}{HTML}{56B4E9}     % 95% band
\definecolor{oiVerm}{HTML}{D55E00}    % one sample

\newcommand{\qepfpanel}[3][]{% #1 = extra axis options, #2 = file stem, #3 = title
  \nextgroupplot[title={#3},#1]
  \addplot[name path=hi, draw=none, forget plot] table[x=u, y=hi, col sep=comma] {qepf_#2_MCband.csv};
  \addplot[name path=lo, draw=none, forget plot] table[x=u, y=lo, col sep=comma] {qepf_#2_MCband.csv};
  \addplot[oiSky, fill opacity=0.30] fill between[of=hi and lo];
  \addplot[oiGreen, solid, line width=1.6pt] table[x=u, y=P, col sep=comma] {qepf_#2_Theoretical.csv};
  \addplot[oiBlue, densely dotted, line width=1.4pt] table[x=u, y=mean, col sep=comma] {qepf_#2_MCband.csv};
  \addplot[oiVerm, dashed, line width=1.1pt] table[x=u, y=P, col sep=comma] {qepf_#2_Empirical.csv};
}
